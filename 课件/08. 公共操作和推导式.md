# 5.5 enumerate() 枚举函数  
# enumerate(可遍历的对象， start默认为0)  
# 返回结果是元组  
list1 = ["a", "b", "c", "d", "e", "f"]  
for i in enumerate(list1):  
    print(i)
# (0, 'a')
# (1, 'b')
# (2, 'c')
# (3, 'd')
# (4, 'e')
# (5, 'f')
for i in enumerate(list1, start = 3):  # start可以改变序列中数据对应的编号
    print(i)
# (3, 'a')  
# (4, 'b')  
# (5, 'c')  
# (6, 'd')  
# (7, 'e')  
# (8, 'f')

## 2.推导式
```python
# 2.1 列表推导式
# 需求：生成1-10数字  
# while循环  
list1 = []  
i = 1  
while i <= 10:  
    list1.append(i)  
    i += 1  
print(list1)  
# for循环  
list1 = []  
for i in range(1, 11):  
    list1.append(i)  
print(list1)  
# 列表推导式  
list1 = [i for i in range(1, 11)]  
print(list1)

# 2.2 带if语句的列表推导式
# 需求：生成1-10的单数列表  
list1 = [i for i in range(1, 11) if i % 2 != 0]  
print(list1)

# 2.3 多个for循环实现列表推导式
# 需求：[(1, 0), (1, 1), (1, 2), (2, 0), (2, 1), (2, 2)]
# for循环嵌套  
list1 = []  
for i in range(1, 3):  
    for j in range(3):  
        list1.append((i,j))  
print(list1)  
# 列表推导式 
list1 = [(i, j) for i in range(1, 3) for j in range(3)]  
print(list1)

# 2.4 字典推导式
# 2.4.1 字典推导式创建字典
# 需求：key为1-5，value为key的平方
dict1 = {i : i ** 2 for i in range(1, 6)}  
print(dict1)    # {1: 1, 2: 4, 3: 9, 4: 16, 5: 25}

# 2.4.2 将两个列表合并为字典  
list1 = ["name", "age", "score"]  
list2 = ["Yuan", 22, 97]  
dict1 = {list1[i] : list2[i] for i in range(len(list1))}  
print(dict1)    # {'name': 'Yuan', 'age': 22, 'score': 97}
```
