# 12.闭包与装饰器

## 1.闭包
```python
# 1.闭包
# 1.1 定义：函数嵌套的前提下，内部函数使用了外部函数的变量并且外部函数返回了内部函数，我们把这个使用外部函数变量的内部函数称为闭包。
# 1.2 作用：保存外部函数内的变量
# 1.3 构成条件：(1)函数嵌套
#              (2)内部函数使用了外部函数的变量或参数
#              (3)外部函数返回了内部函数

# 1.4 构成闭包
def func_out():
    x = 100
    def func_inner(y):
        result = x + y
        print(result)
    return func_inner   # 不加括号，加了表示运行该函数

# 1.5 使用闭包
result = func_out() # 即result = func_inner
result(1)   # 101

# 1.6 应用实例
# 需求：根据配置信息使用闭包实现不同人的对话信息
# 实现对话：
# 张三：到北京了吗？ 李四：已经到了，放心吧。
# 实现步骤：(1)定义外部函数 接收不同的配置信息参数（人名）
#          (2)定义内部函数 接收对话信息参数
#          (3)在内部参数里面把配置信息和对话信息进行拼接输出
def func_out(name):
    def func_inner(message):
        result = f"{name}:{message}"
        print(result)
    return func_inner

zhans = func_out("张三")
lis = func_out("李四")

zhans("到北京了吗？")
lis("已经到了，放心吧。")

# 思考：
# 使用闭包比普通函数好在哪里？
# 如果使用普通函数，每次输入信息都需要重复输入"张三""李四"，而使用闭包只需要每次输入他们的话。
# 那为什么不写成两个分属于张三和李四的函数呢？
# 两个人还好，如果是两百个呢？难道要写两百个函数吗？使用闭包，只需定义一次外部函数，就可以创建任意多个具有不同配置信息的闭包。
# 所以，闭包的作用：保存外部函数的变量，使内部函数在外部函数执行结束后，仍然能够访问这些变量。在本例中，保存的变量是 name
```

## 2.装饰器
```python
# 2.1 定义
# 给已有函数增加额外功能的函数，本质就是一个闭包函数

# 2.2 功能特点
# (1)不修改已有函数的源代码
# (2)不修改已有函数的调用方式
# (3)给已有函数增加额外的功能

# 2.3 装饰器的语法糖写法
#   @装饰器名字

# 2.4 应用
# 需求：原函数comment，需要新功能时，不想直接修改原函数，希望使用闭包进行装饰
def out(func):
    def inner():
        func()
        print("成功")
    return inner

def comment():
    print("输入")
    # print("成功")

new_func = out(comment)
new_func()

# 2.4 装饰器的语法糖写法
#   @装饰器名字

def out(func):
    def inner():
        func()
        print("成功")
    return inner

@out    # 放在要修饰的函数的上方，相当于comment = out(comment)

def comment():
    print("输入")
    # print("成功")

comment()   #输入 \n 成功

# 2.5 装饰器的使用
# 2.5.1 场景1：统计函数执行的时间

import time
# 获取的时间是距离1970年1月1日 0:00:00的时间，以秒为单位

def out(func):
    def inner():
        # 获取函数执行前的时间
        begin = time.time()
        func()
        # 获取函数执行后的时间
        end = time.time()
        # 两时间相减
        spend = end - begin
        print("函数执行用时：", spend)
    return inner

@out

def work():
    for i in range(10000):
        print(i)

work()  # ... 函数执行用时： 0.21831583976745605

# 2.5.2 通用装饰器：装饰任意的函数——内部函数类型和被装饰函数保持一致
# (1)装饰带参数的函数
# 装饰器
def out(func):
    def inner(a, b):
        print("正在执行加法运算")
        func(a, b)
    return inner

@out
def add_num(n1, n2):
    result = n1 + n2
    print(result)

add_num(1, 2)   # 正在执行加法运算 /n 3

# 解析
# @out相当于拿inner()替代了原add_num()，而add_num()传入了两个参数，那么inner()必须能接收这两个参数，故inner(a, b)
# func()相当于add_num()，当然也要传入两个参数
# 参数传递过程：add_num(1,2)→inner(a=1,b=2)→func(a=1,b=2)→add_num(n1=1,n2=2)

# (2)装饰带有参数和返回值的函数
# 装饰器
def out(func):
    def inner(a, b):
        print("正在执行加法运算")
        result1 = func(a, b)
        return result1
    return inner

@out
def add_num(n1, n2):
    result = n1 + n2
    return result

result2 = add_num(1, 2)
print(result2)  # 正在执行加法运算 /n 3

# (3)装饰带有不定长参数和返回值的函数
# 装饰器——此时的装饰器就是通用装饰器了
def out(func):
    def inner(*args, **kwargs):
        print("正在执行加法运算")
        result1 = func(*args, **kwargs)
        return result1
    return inner

@out
def add_num(*args, **kwargs):
    result = 0
    for value in args:
        result += value
    for value in kwargs.values():
        result += value
    return result

result2 = add_num(1, 2, age=22)
print(result2)  # 正在执行加法运算 /n 25

# 2.5.3 多个装饰器的使用
# 要实现：<div><p>"慨当以慷，忧思难忘"</p><div>

# 装饰器1：作用p标签
def make_p(func):
    def inner():
        result = "<p>" + func() + "</p>"
        return result
    return inner

# 装饰器2：作用div标签
def make_div(func):
    def inner():
        result = "<div>" + func() + "<div>"
        return result
    return inner

# 注意写法，一层层装饰
@ make_div  # 装饰已经被make_p装饰过的content()
@ make_p    # 装饰content()
def content():
    return "慨当以慷，忧思难忘"

result = content()
print(result)   # <div><p>慨当以慷，忧思难忘</p><div>

# 带有参数的装饰器
def return_decorate(flag):
    def decorate(func):
        def inner(a, b):
            if flag == "+":
                print("正在相加")
            elif flag == "-":
                print("正在相减")
            func(a, b)
        return inner
    return decorate

@return_decorate("+")
def add_num(a, b):
    result = a + b
    print(result)
add_num(1, 2)   # 正在相加 /n 3

@return_decorate("-")
def sub_num(a, b):
    result = a - b
    print(result)
sub_num(1, 2)   # # 正在相减 /n -1
```
